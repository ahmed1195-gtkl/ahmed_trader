rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    
    // وظيفة للتحقق مما إذا كان المستخدم هو أدمن
    function isAdmin() {
      return request.auth != null && (
        request.auth.token.email == 'mchokri100@gmail.com' || 
        request.auth.token.email == 'ahmed1195@gmail.com' ||
        get(/databases/$(database)/documents/users/$(request.auth.uid)).data.isAdmin == true
      );
    }

    // وظيفة للتحقق مما إذا كان المستخدم مسجل دخول
    function isSignedIn() {
      return request.auth != null;
    }

    // قواعد مجموعة المستخدمين
    match /users/{userId} {
      allow read: if isSignedIn();
      allow create: if isSignedIn() && request.auth.uid == userId;
      allow update: if isAdmin() || (isSignedIn() && request.auth.uid == userId);
      allow delete: if isAdmin();
    }

    // قواعد صفقات البوت
    match /bot_trades/{tradeId} {
      allow read: if true;
      allow create: if true; 
      allow update, delete: if isAdmin();
    }

    // قواعد إعدادات البوت
    match /bot_settings/{settingId} {
      allow read: if true;
      allow write: if true; 
    }

    // قواعد المنشورات
    match /posts/{postId} {
      allow read: if true;
      allow create: if isAdmin();
      allow update: if isSignedIn();
      allow delete: if isAdmin();
    }

    // قواعد منشورات الأدمن (للتوافق)
    match /admin_posts/{postId} {
      allow read: if true;
      allow write: if isAdmin();
    }

    // قواعد التعليقات
    match /comments/{commentId} {
      allow read: if true;
      allow create: if isSignedIn();
      allow update, delete: if isAdmin() || (isSignedIn() && resource.data.userId == request.auth.uid);
    }

    // قواعد الرسائل القديمة (للتوافق)
    match /messages/{messageId} {
      allow read: if isAdmin() || (isSignedIn() && resource.data.userId == request.auth.uid);
      allow create: if isSignedIn() && request.resource.data.userId == request.auth.uid;
      allow update, delete: if isAdmin();
    }

    // قواعد المحادثات الجديدة (نظام Messenger)
    match /conversations/{conversationId} {
      allow read: if isAdmin() || (isSignedIn() && request.auth.uid in resource.data.participants);
      allow create: if isSignedIn() && request.auth.uid in request.resource.data.participants;
      allow update: if isSignedIn() && request.auth.uid in resource.data.participants;
      allow delete: if isAdmin();

      // قواعد الرسائل داخل المحادثة
      match /messages/{messageId} {
        allow read: if isAdmin() || (isSignedIn() && request.auth.uid in get(/databases/$(database)/documents/conversations/$(conversationId)).data.participants);
        allow create: if isSignedIn() && request.auth.uid in get(/databases/$(database)/documents/conversations/$(conversationId)).data.participants && request.resource.data.senderId == request.auth.uid;
        allow update: if false;
        allow delete: if isAdmin() || (isSignedIn() && resource.data.senderId == request.auth.uid);
      }
    }

    // قواعد إعدادات الموقع
    match /site_settings/{settingId} {
      allow read: if isSignedIn();
      allow write: if isAdmin();
    }

    // قواعد سجل النشاط
    match /activity_logs/{logId} {
      allow read: if isAdmin();
      allow create: if isAdmin();
      allow update, delete: if false;
    }

    // قواعد الوسطاء
    match /brokers/{brokerId} {
      allow read: if true;
      allow write: if isAdmin();
    }

    // قواعد طلبات الصداقة
    match /friend_requests/{requestId} {
      allow read: if isSignedIn() && (resource.data.senderId == request.auth.uid || resource.data.receiverId == request.auth.uid);
      allow create: if isSignedIn() && request.resource.data.senderId == request.auth.uid;
      allow update: if isSignedIn() && resource.data.receiverId == request.auth.uid;
      allow delete: if isSignedIn() && (resource.data.senderId == request.auth.uid || resource.data.receiverId == request.auth.uid);
    }
  }
}

ملاحظات مهمة:
========================================

1. انسخ الكود أعلاه بالكامل
2. اذهب إلى Firebase Console
3. اختر مشروعك
4. اذهب إلى Firestore Database > Rules
5. احذف القواعد القديمة بالكامل
6. الصق القواعد الجديدة
7. اضغط "Publish" أو "نشر"

========================================
المجموعات المستخدمة في Firebase:
========================================

✅ users - بيانات المستخدمين
✅ posts - المنشورات
✅ comments - التعليقات
✅ conversations - المحادثات (جديد)
   └── messages - الرسائل داخل كل محادثة (subcollection)
✅ friend_requests - طلبات الصداقة
✅ bot_trades - صفقات البوت
✅ bot_settings - إعدادات البوت
✅ brokers - الوسطاء
✅ site_settings - إعدادات الموقع
✅ activity_logs - سجل النشاط
✅ messages - الرسائل القديمة (للتوافق)
✅ admin_posts - منشورات الأدمن القديمة (للتوافق)
