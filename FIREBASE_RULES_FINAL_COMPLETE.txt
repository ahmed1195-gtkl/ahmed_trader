rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    
    // ======================================================
    // وظائف المساعدة الأساسية (Helper Functions)
    // ======================================================

    // التحقق مما إذا كان المستخدم هو أدمن
    function isAdmin() {
      return request.auth != null && (
        request.auth.token.email.toLowerCase() in ['mchokri100@gmail.com', 'ahmed1195@gmail.com'] ||
        get(/databases/$(database)/documents/users/$(request.auth.uid)).data.isAdmin == true
      );
    }

    // التحقق مما إذا كان المستخدم مسجل دخول
    function isSignedIn() {
      return request.auth != null;
    }

    // التحقق من ملكية المستند
    function isOwner(userId) {
      return isSignedIn() && request.auth.uid == userId;
    }

    // ======================================================
    // قواعد المجموعات (Collection Rules)
    // ======================================================

    // 1. المستخدمين (Users)
    match /users/{userId} {
      allow read: if isSignedIn();
      allow create: if isSignedIn() && request.auth.uid == userId;
      allow update: if isAdmin() || isOwner(userId);
      allow delete: if isAdmin();
    }

    // 2. الكورسات (Courses)
    match /courses/{courseId} {
      allow read: if true;
      allow write: if isAdmin();
    }

    // 3. التحديات (Challenges)
    match /challenges/{challengeId} {
      allow read: if true;
      // السماح للمستخدمين بإنشاء تحديات (مطلوب في TradingChallengeTest)
      allow create: if isSignedIn();
      // السماح بتحديث التحدي (مثل زيادة عدد المشاركين) عند الانضمام
      allow update: if isSignedIn();
      allow delete: if isAdmin();
    }

    // 4. المشاركين في التحديات (Challenge Participants)
    match /challenge_participants/{participantId} {
      allow read: if isSignedIn();
      allow create: if isSignedIn() && request.resource.data.userId == request.auth.uid;
      allow update: if isAdmin() || (isSignedIn() && resource.data.userId == request.auth.uid);
      allow delete: if isAdmin();
    }

    // 5. صفقات التحديات (Challenge Trades)
    match /challenge_trades/{tradeId} {
      allow read: if isSignedIn();
      allow create: if isSignedIn() && request.resource.data.userId == request.auth.uid;
      allow update: if isAdmin() || (isSignedIn() && resource.data.userId == request.auth.uid);
      allow delete: if isAdmin() || (isSignedIn() && resource.data.userId == request.auth.uid);
    }

    // 6. الفرق (Teams)
    match /teams/{teamId} {
      allow read: if isSignedIn();
      allow create: if isSignedIn();
      allow update: if isSignedIn() && (request.auth.uid in resource.data.members || isAdmin());
      allow delete: if isAdmin() || (isSignedIn() && resource.data.leaderId == request.auth.uid);
    }

    // 7. دعوات الفرق (Team Invites)
    match /team_invites/{inviteId} {
      allow read: if isSignedIn();
      allow create: if isSignedIn();
      allow update: if isSignedIn();
      allow delete: if isAdmin();
    }

    // 8. دردشة الفريق (Team Chat)
    match /team_chat/{messageId} {
      allow read: if isSignedIn();
      allow create: if isSignedIn();
      allow update, delete: if isAdmin() || (isSignedIn() && resource.data.userId == request.auth.uid);
    }

    // 9. الرسائل والدردشة الخاصة (Messages & Conversations)
    match /messages/{messageId} {
      allow read: if isAdmin() || (isSignedIn() && resource.data.userId == request.auth.uid);
      allow create: if isSignedIn();
      allow update, delete: if isAdmin();
    }

    match /conversations/{conversationId} {
      allow read: if isAdmin() || (isSignedIn() && request.auth.uid in resource.data.participants);
      allow create: if isSignedIn() && request.auth.uid in request.resource.data.participants;
      allow update: if isSignedIn() && request.auth.uid in resource.data.participants;
      allow delete: if isAdmin();

      match /messages/{messageId} {
        allow read: if isAdmin() || (isSignedIn() && request.auth.uid in get(/databases/$(database)/documents/conversations/$(conversationId)).data.participants);
        allow create: if isSignedIn() && request.auth.uid in get(/databases/$(database)/documents/conversations/$(conversationId)).data.participants && request.resource.data.senderId == request.auth.uid;
        allow update: if false;
        allow delete: if isAdmin() || (isSignedIn() && resource.data.senderId == request.auth.uid);
      }
    }

    // 10. المنشورات والتعليقات (Posts & Comments)
    match /posts/{postId} {
      allow read: if true;
      allow create: if isAdmin();
      allow update: if isSignedIn();
      allow delete: if isAdmin();
    }

    match /comments/{commentId} {
      allow read: if true;
      allow create: if isSignedIn();
      allow update, delete: if isAdmin() || (isSignedIn() && resource.data.userId == request.auth.uid);
    }

    // 11. طلبات الصداقة (Friend Requests)
    match /friend_requests/{requestId} {
      allow read: if isSignedIn() && (resource.data.senderId == request.auth.uid || resource.data.receiverId == request.auth.uid);
      allow create: if isSignedIn() && request.resource.data.senderId == request.auth.uid;
      allow update: if isSignedIn() && (resource.data.receiverId == request.auth.uid || resource.data.senderId == request.auth.uid);
      allow delete: if isSignedIn() && (resource.data.senderId == request.auth.uid || resource.data.receiverId == request.auth.uid);
    }

    // 12. حسابات التداول (Demo Accounts)
    match /demo_accounts/{accountId} {
      allow read: if isAdmin() || (isSignedIn() && resource.data.userId == request.auth.uid);
      allow write: if isAdmin() || (isSignedIn() && request.resource.data.userId == request.auth.uid);
    }

    // 13. لوحات الصدارة (Leaderboards)
    match /global_leaderboard/{userId} {
      allow read: if true;
      allow write: if isAdmin() || (isSignedIn() && request.auth.uid == userId);
    }

    match /challenge_leaderboards/{leaderboardId} {
      allow read: if true;
      allow write: if isAdmin() || isSignedIn();
    }

    // 14. إعدادات الموقع وسجلات النشاط (Site Settings & Logs)
    match /site_settings/{settingId} {
      allow read: if true;
      allow write: if isAdmin();
    }

    match /activity_logs/{logId} {
      allow read: if isAdmin();
      allow create: if isSignedIn();
      allow update, delete: if false;
    }

    match /balance_audit_log/{auditId} {
      allow read: if isAdmin() || (isSignedIn() && resource.data.userId == request.auth.uid);
      allow create: if isSignedIn();
      allow update, delete: if false;
    }

    match /notifications/{notificationId} {
      allow read: if isSignedIn() && resource.data.userId == request.auth.uid;
      allow create: if isAdmin() || isSignedIn();
      allow update: if isSignedIn() && resource.data.userId == request.auth.uid;
      allow delete: if isSignedIn() && resource.data.userId == request.auth.uid;
    }

    // 15. صفقات البوت وإعداداته (Bot Trades & Settings)
    match /bot_trades/{tradeId} {
      allow read: if true;
      allow create: if true; 
      allow update, delete: if isAdmin();
    }

    match /bot_settings/{settingId} {
      allow read: if true;
      allow write: if true; 
    }

    // 16. الوسطاء (Brokers)
    match /brokers/{brokerId} {
      allow read: if true;
      allow write: if isAdmin();
    }
  }
}
