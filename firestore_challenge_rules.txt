// قواعد Firestore الإضافية لنظام التحديات
// يجب إضافة هذه القواعد إلى ملف firestore.rules الموجود

// مجموعة التحديات - يمكن للجميع القراءة، فقط الأدمن يمكنه الكتابة
match /challenges/{challengeId} {
  allow read: if true;
  allow write: if request.auth != null && 
    (get(/databases/$(database)/documents/users/$(request.auth.uid)).data.isAdmin == true ||
     request.auth.token.email.toLowerCase() in ['mchokri100@gmail.com', 'ahmed1195@gmail.com']);
}

// مجموعة المشاركين في التحديات
match /challenge_participants/{participantId} {
  allow read: if request.auth != null;
  allow create: if request.auth != null && request.auth.uid == request.resource.data.userId;
  allow update: if request.auth != null && 
    (request.auth.uid == resource.data.userId || 
     get(/databases/$(database)/documents/users/$(request.auth.uid)).data.isAdmin == true);
  allow delete: if request.auth != null && 
    get(/databases/$(database)/documents/users/$(request.auth.uid)).data.isAdmin == true;
}

// مجموعة صفقات التحديات
match /challenge_trades/{tradeId} {
  allow read: if request.auth != null;
  allow create: if request.auth != null && request.auth.uid == request.resource.data.userId;
  allow update: if request.auth != null && request.auth.uid == resource.data.userId;
  allow delete: if request.auth != null && 
    (request.auth.uid == resource.data.userId ||
     get(/databases/$(database)/documents/users/$(request.auth.uid)).data.isAdmin == true);
}

// مجموعة لوحات الصدارة
match /challenge_leaderboards/{leaderboardId} {
  allow read: if true;
  allow write: if false; // يتم التحديث فقط من خلال Cloud Functions أو الأدمن
}
